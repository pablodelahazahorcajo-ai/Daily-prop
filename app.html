<!DOCTYPE html>

<html lang="es">

<head>

    <meta charset="UTF-8">

    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>Prop Trading Dashboard</title>

    <script src="https://cdn.tailwindcss.com"></script>

    <script crossorigin src="https://unpkg.com/react@18/umd/react.production.min.js"></script>

    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>

</head>

<body class="bg-gradient-to-br from-slate-50 to-slate-100">

    <div id="root"></div>

    <script>

        const { useState, useEffect } = React;

        const APPS_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbw5subAalhQuoXGDuPG1-m031iAPm-Oz4yl20G_9wxTzgA_ExcohYpC1XiV5q-yw4W-/exec';

        const API = {

            async getData() {

                try {

                    const response = await fetch(`${APPS_SCRIPT_URL}?action=getData`);

                    return await response.json();

                } catch (error) {

                    console.error('Error:', error);

                    return null;

                }

            },

            async saveEntry(trader, entry) {

                try {

                    const response = await fetch(APPS_SCRIPT_URL, {

                        method: 'POST',

                        body: JSON.stringify({ action: 'saveEntry', trader, entry })

                    });

                    return await response.json();

                } catch (error) {

                    console.error('Error:', error);

                    return null;

                }

            },

            async executeDaily() {

                try {

                    const response = await fetch(APPS_SCRIPT_URL, {

                        method: 'POST',

                        body: JSON.stringify({ action: 'executeDaily' })

                    });

                    return await response.json();

                } catch (error) {

                    console.error('Error:', error);

                    return null;

                }

            }

        };


        function App() {

            const [data, setData] = useState({ traders: {} });

            const [loading, setLoading] = useState(true);

            const [activeTab, setActiveTab] = useState('pablo');

            const [isExecuting, setIsExecuting] = useState(false);


            useEffect(() => {

                loadData();

            }, []);


            const loadData = async () => {

                const result = await API.getData();

                if (result) setData(result);

                setLoading(false);

            };


            const handleExecuteDaily = async () => {

                setIsExecuting(true);

                await API.executeDaily();

                await loadData();

                setIsExecuting(false);

            };


            const handleSaveEntry = async (trader, entry) => {

                await API.saveEntry(trader, entry);

                await loadData();

            };


            if (loading) return <div className="flex items-center justify-center min-h-screen"><p>Cargando...</p></div>;


            return (

                <div className="min-h-screen bg-gradient-to-br from-slate-50 to-slate-100">

                    <header className="border-b border-slate-200 bg-white shadow-sm">

                        <div className="max-w-7xl mx-auto px-4 py-4">

                            <div className="flex justify-between items-center">

                                <h1 className="text-3xl font-bold">📊 Prop Trading Dashboard</h1>

                                <button onClick={handleExecuteDaily} disabled={isExecuting} className="px-4 py-2 bg-green-600 text-white rounded-lg">

                                    {isExecuting ? 'Ejecutando...' : 'Ejecutar Diario'}

                                </button>

                            </div>

                        </div>

                    </header>


                    <main className="max-w-7xl mx-auto px-4 py-8">

                        <div className="flex gap-2 mb-6">

                            {['pablo', 'javi', 'guillem', 'alex'].map(tab => (

                                <button key={tab} onClick={() => setActiveTab(tab)} className={`px-4 py-2 font-bold border-b-2 ${activeTab === tab ? 'border-blue-600 text-blue-600' : 'border-transparent'}`}>

                                    {tab.toUpperCase()}

                                </button>

                            ))}

                        </div>


                        <div className="bg-white rounded-lg border border-slate-200 p-6">

                            <h2 className="text-2xl font-bold mb-4">{activeTab.toUpperCase()}</h2>

                            <table className="w-full text-sm">

                                <thead>

                                    <tr className="border-b bg-slate-50">

                                        <th className="px-4 py-2 text-left">Fecha</th>

                                        <th className="px-4 py-2 text-left">Empresa</th>

                                        <th className="px-4 py-2 text-left">Saldo</th>

                                        <th className="px-4 py-2 text-left">Fase</th>

                                        <th className="px-4 py-2 text-left">P&L</th>

                                    </tr>

                                </thead>

                                <tbody>

                                    {(data.traders[activeTab.toUpperCase()]?.brokers || []).map((entry, idx) => (

                                        <tr key={idx} className="border-b">

                                            <td className="px-4 py-2">{entry.date}</td>

                                            <td className="px-4 py-2">{entry.company}</td>

                                            <td className="px-4 py-2">${entry.saldo}</td>

                                            <td className="px-4 py-2">{entry.fase}</td>

                                            <td className="px-4 py-2">${entry.dailyProfit}</td>

                                        </tr>

                                    ))}

                                </tbody>

                            </table>

                        </div>

                    </main>

                </div>

            );

        }


        const root = ReactDOM.createRoot(document.getElementById('root'));

        root.render(<App />);

    </script>

</body>

</html>
